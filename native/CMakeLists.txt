cmake_minimum_required(VERSION 3.22)
project(xpn VERSION 0.1.2 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(OpenSSL 3.0 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LWS REQUIRED IMPORTED_TARGET libwebsockets>=4.3)
pkg_check_modules(JSON REQUIRED IMPORTED_TARGET jansson>=2.13)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl>=7.85)
find_package(Threads REQUIRED)
add_library(xpn STATIC src/core.c src/transport.c src/standby.c src/profile.c src/certificate.c)
target_include_directories(xpn PUBLIC include PRIVATE src)
target_link_libraries(xpn PUBLIC OpenSSL::SSL OpenSSL::Crypto PkgConfig::LWS PkgConfig::JSON PkgConfig::CURL Threads::Threads)
target_compile_definitions(xpn PRIVATE _POSIX_C_SOURCE=200809L)
if(APPLE)
  target_compile_definitions(xpn PRIVATE _DARWIN_C_SOURCE)
else()
  target_compile_definitions(xpn PRIVATE _DEFAULT_SOURCE)
endif()
target_compile_options(xpn PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-missing-field-initializers)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(xpnd src/linux.c src/privilege_linux.c)
  target_link_libraries(xpnd xpn)
  target_compile_options(xpnd PRIVATE -Wall -Wextra -Wpedantic -Werror)
  install(TARGETS xpnd RUNTIME DESTINATION bin)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_core tests/test_core.c)
  target_include_directories(test_core PRIVATE src)
  target_link_libraries(test_core xpn)
  target_compile_options(test_core PRIVATE -UNDEBUG)
  add_test(NAME core COMMAND test_core)
  add_executable(test_options tests/test_options.c)
  target_include_directories(test_options PRIVATE src)
  target_link_libraries(test_options xpn)
  target_compile_options(test_options PRIVATE -UNDEBUG)
  add_test(NAME encryption_options COMMAND test_options)
  add_executable(test_standby tests/test_standby.c)
  target_include_directories(test_standby PRIVATE src)
  target_link_libraries(test_standby xpn)
  target_compile_options(test_standby PRIVATE -UNDEBUG)
  add_test(NAME standby COMMAND test_standby)
  add_executable(test_certificate tests/test_certificate.c)
  target_include_directories(test_certificate PRIVATE src)
  target_link_libraries(test_certificate xpn)
  target_compile_options(test_certificate PRIVATE -UNDEBUG)
  add_test(NAME certificate COMMAND test_certificate)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(test_privilege tests/test_privilege.c src/privilege_linux.c)
    target_include_directories(test_privilege PRIVATE src)
    target_link_libraries(test_privilege Threads::Threads)
    target_compile_options(test_privilege PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME privilege COMMAND test_privilege)
    # Disabling process dumps is intentional; LSan uses a ptrace-based scanner.
    set_tests_properties(privilege PROPERTIES ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0")
  endif()
  add_executable(test_transport tests/test_transport.c)
  target_include_directories(test_transport PRIVATE src)
  target_link_libraries(test_transport xpn)
  target_compile_options(test_transport PRIVATE -UNDEBUG)
endif()
